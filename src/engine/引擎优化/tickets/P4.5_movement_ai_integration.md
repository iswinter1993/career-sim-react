# Phase 4.5: 三层AI集成到球员移动 — strategy + role + trait 影响 intentPOS

**Phase**: 4 (Three-Layer AI)
**优先级**: P2
**阻塞**: P1.4 (playerMovement增强), P4.1-P4.3
**被阻塞**: P4.6 (MatchView集成)

## 描述

将 TeamStrategy + PlayerRole + PlayerTraits 集成到 `playerMovement.js` 的意图位置(intentPOS)计算和移动决策中，使球队风格和球员角色影响每个球员的无球跑动行为。

## 背景

当前 `playerMovement.js` 的移动逻辑仅基于位置分组（P1.4已做）。本ticket在其上叠加三层AI:

**Layer 1 (TeamStrategy)**: 影响整体的移动参数
- 防线高度 → 调整所有防守球员的 Y 基准
- 宽度 → 调整所有边路球员的 X 偏移
- 压迫强度 → 影响 attacking intentPOS 时的压缩因子
- 流动度 → 允许一定程度的 position swapping

**Layer 2 (PlayerRole)**: 影响个体移动倾向
- `forwardRuns` → 进攻时前插幅度因子
- `holdPosition` → 防守时位置保持因子
- `stayWide` → 宽度保持因子（边路角色更高）
- `cutInside` → 内切倾向（内锋等角色）
- `roamFromPosition` → 自由移动幅度

**Layer 3 (PlayerTraits)**: 最终微调
- `cuts_inside_from_both_flanks` → 覆盖 stayWide
- `gets_forward_whenever_possible` → 覆盖 forwardRuns
- 等

1. **创建 `getMovementModifiers(player, teamStrategy, matchState)` 函数**:

```javascript
import { getRoleModifier, evaluateTraits } from './tactics.js';
import { getPositionGroup } from './positionGroup.js';

/**
 * Calculate the movement modifiers for a player from all three AI layers.
 * Called by decideMovement() / setIntentPosition().
 *
 * @returns {{forwardRuns: number, holdPosition: number, stayWide: number, 
 *            cutInside: number, roamFromPosition: number, closingDown: number,
 *            staminaDrain: number}}
 */
export function getMovementModifiers(player, teamStrategy, matchState) {
  // Start with default (no modifier)
  const modifiers = {
    forwardRuns: 1.0,
    holdPosition: 1.0,
    stayWide: 1.0,
    cutInside: 1.0,
    roamFromPosition: 1.0,
    closingDown: 1.0,
    staminaDrain: 1.0,
  };
  
  // Layer 1: Team Strategy
  if (teamStrategy) {
    if (teamStrategy._widthMultiplier) {
      modifiers.stayWide *= teamStrategy._widthMultiplier;
    }
    if (teamStrategy._pressingMultiplier) {
      modifiers.closingDown *= teamStrategy._pressingMultiplier;
    }
    if (teamStrategy._fluidityFactor !== undefined) {
      modifiers.roamFromPosition += teamStrategy._fluidityFactor;
      modifiers.holdPosition -= teamStrategy._fluidityFactor * 0.5;
    }
  }
  
  // Layer 2: Player Role
  if (player.role) {
    const role = getRoleModifier(player.role);
    if (role.movementModifiers) {
      const rm = role.movementModifiers;
      if (rm.forwardRuns !== undefined) modifiers.forwardRuns += rm.forwardRuns;
      if (rm.holdPosition !== undefined) modifiers.holdPosition += rm.holdPosition;
      if (rm.stayWide !== undefined) modifiers.stayWide += rm.stayWide;
      if (rm.cutInside !== undefined) modifiers.cutInside += rm.cutInside;
      if (rm.roamFromPosition !== undefined) modifiers.roamFromPosition += rm.roamFromPosition;
      if (rm.closingDown !== undefined) modifiers.closingDown += rm.closingDown;
      if (rm.staminaDrain !== undefined) modifiers.staminaDrain += rm.staminaDrain;
    }
  }
  
  // Layer 3: Player Traits
  if (player.traits && player.traits.length > 0) {
    const traitEffects = evaluateTraits(player, matchState);
    if (traitEffects.movementModifiers) {
      const tm = traitEffects.movementModifiers;
      if (tm.forwardRuns !== undefined) modifiers.forwardRuns += tm.forwardRuns;
      if (tm.holdPosition !== undefined) modifiers.holdPosition += tm.holdPosition;
      if (tm.stayWide !== undefined) modifiers.stayWide += tm.stayWide;
      if (tm.cutInside !== undefined) modifiers.cutInside += tm.cutInside;
      if (tm.roamFromPosition !== undefined) modifiers.roamFromPosition += tm.roamFromPosition;
      if (tm.closingDown !== undefined) modifiers.closingDown += tm.closingDown;
      if (tm.staminaDrain !== undefined) modifiers.staminaDrain += tm.staminaDrain;
    }
  }
  
  // Clamp to reasonable ranges
  for (const key of Object.keys(modifiers)) {
    modifiers[key] = Math.max(0.1, Math.min(2.0, modifiers[key]));
  }
  
  return modifiers;
}
```

2. **集成到 `decideMovement()` / `setIntentPosition()`**:

```javascript
// 在 setAttackRelativePos() 中:
export function setAttackRelativePos(ball, player, homeAttacking, pitch, teamStrategy) {
  // ...existing position group differentiation (P1.4)...
  
  // 新：叠加三层AI移动修正
  const modifiers = getMovementModifiers(player, teamStrategy, {
    ballPossession: homeAttacking ? 'home' : 'away',
    ball,
  });
  
  // Apply forwardRuns modifier
  pushFactor *= modifiers.forwardRuns;
  
  // Apply roamFromPosition — adds random noise to intentPOS
  if (modifiers.roamFromPosition > 1.0) {
    const roamNoise = (modifiers.roamFromPosition - 1.0) * 60;
    intentPOS[0] += (Math.random() - 0.5) * roamNoise;
    intentPOS[1] += (Math.random() - 0.5) * roamNoise * 0.5;
  }
  
  // Apply stayWide/cutInside to X coordinate
  if (modifiers.stayWide > 1.0) {
    // Pull wider — increase X offset from center
    const centerX = pitch.x / 2;
    intentPOS[0] += (intentPOS[0] - centerX) * (modifiers.stayWide - 1.0) * 0.5;
  }
  if (modifiers.cutInside > 1.0) {
    // Pull inside — decrease X offset from center
    const centerX = pitch.x / 2;
    intentPOS[0] -= (intentPOS[0] - centerX) * (modifiers.cutInside - 1.0) * 0.5;
  }
  
  return intentPOS;
}

// 在 setDefenceRelativePos() 中:
export function setDefenceRelativePos(ball, player, homeDefending, pitch, teamStrategy) {
  // ...existing position group differentiation (P1.4)...
  
  const modifiers = getMovementModifiers(player, teamStrategy, {
    ballPossession: homeDefending ? 'opponent' : 'home',
    ball,
  });
  
  // Apply holdPosition modifier — higher = tighter to originPOS
  compressFactor *= (2.0 - modifiers.holdPosition); // invert: hold=1→compress=1, hold=2→compress=0
  compressFactor = Math.max(0.05, Math.min(0.5, compressFactor));
  
  return intentPOS;
}
```
