3. **换人面板 (SubstitutionPanel)**:

   ```jsx
   function InMatchSubPanel({ matchDetails, side, bench, subTracker, onConfirm }) {
     const [selectedOut, setSelectedOut] = useState(null);
     const [selectedIn, setSelectedIn] = useState(null);
     
     // 左边：场上球员（可被换下）
     const fieldPlayers = matchDetails[`${side}Team`].players
       .filter(p => p.currentPOS[0] !== 'NP'); // 不可选红牌球员
     
     // 右边：替补席球员（可换上）
     const benchPlayers = bench.filter(
       b => !subTracker.subbedOutPlayerIDs.has(b.playerID) // 不可选已上场的
     );
     
     return (
       <div className="sub-panel">
         <div className="sub-panel-left">
           <h3>换下</h3>
           {fieldPlayers.map(p => (
             <SubPlayerCard key={p.playerID} player={p}
               selected={selectedOut === p.playerID}
               onSelect={() => setSelectedOut(p.playerID)} />
           ))}
         </div>
         <div className="sub-panel-right">
           <h3>换上</h3>
           {benchPlayers.map(p => (
             <SubPlayerCard key={p.playerID} player={p}
               selected={selectedIn === p.playerID}
               onSelect={() => setSelectedIn(p.playerID)} />
           ))}
         </div>
         <button onClick={() => onConfirm(selectedOut, selectedIn)}
           disabled={!selectedOut || !selectedIn}>
           确认换人
         </button>
       </div>
     );
   }
   ```

4. **handleSubstitute 升级**:

   ```jsx
   function handleSubstitute(side, playerOutID, playerIn) {
     const newMd = applySubstitutionV2(
       matchDetails, side, playerOutID, playerIn, subTracker, currentMinute
     );
     if (newMd.success) {
       setMatchDetails(newMd.matchDetails);
       setSubTracker(newMd.subTracker);
     }
   }
   
   function handleTacticalChange(side, newFormation, subs) {
     const newMd = applyFormationChangeWithSubs(
       matchDetails, side, subs, newFormation, subTracker, currentMinute
     );
     if (newMd.success) {
       setMatchDetails(newMd.matchDetails);
       setSubTracker(newMd.subTracker);
       setPendingFormation(null);
       setSubQueue([]);
     }
   }
   ```
