**Phase**: 5 (FM风格评分与统计)
**优先级**: P3
**阻塞**: P1.1 (actions.js的12位置支持)
**被阻塞**: P5.3 (赛后报告)

## 描述

实现FM风格的详细比赛统计追踪系统。在比赛过程中实时记录每个球员的各类统计数据，为赛后评分和报告提供数据基础。

## 背景

FM的赛后统计包括但不限于:
- 射门（总数/射正/射偏/被封堵）、传球（完成/尝试/成功率）、抢断（成功/尝试/成功率）
- 拦截、解围、犯规、被犯规、越位、黄牌、红牌
- 跑动距离、冲刺次数、关键传球、头球成功
- GK统计: 扑救、出击、高空球接住、拳击、丢球

当前引擎在迭代中执行动作，但数据没有被系统性地追踪和归因到具体球员。

```javascript
/**
 * Player-level match statistics tracker.
 * Created once per match, updated each time a tracked event occurs.

export function createMatchStatsTracker() {
  return {
    home: createTeamStatsTracker(),
    away: createTeamStatsTracker(),
    matchEvents: [],     // timeline of key events
    teamStats: {
      home: { possession: 0, shots: 0, shotsOnTarget: 0, corners: 0, fouls: 0, cards: 0 },
      away: { possession: 0, shots: 0, shotsOnTarget: 0, corners: 0, fouls: 0, cards: 0 },
    },
  };
}

function createTeamStatsTracker() {
  return {
    players: {}, // keyed by playerID
    total: createEmptyPlayerStats(),
  };
}

    sweeps: 0,

    offsides: 0,
    assists: 0,
    positionErrors: 0,
    errorsLeadingToGoal: 0,

    minutesPlayed: 0,
    substitutedOff: false,
    substitutedOnMinute: null,
    substitutedOffMinute: null,
  };
}

function getOrCreatePlayerStats(tracker, playerID) {
  if (!tracker.players[playerID]) {
    tracker.players[playerID] = createEmptyPlayerStats();
  }
  return tracker.players[playerID];
}
```

2. **事件记录函数**:

```javascript
/**
 * Record a tracked event during the match.
 *
 * @param {object} tracker — match stats tracker
 * @param {'home'|'away'} side — which team
 * @param {string} playerID — player who performed the action
 * @param {string} eventType — event category
 * @param {object} [detail] — additional event data
 * @param {number} iteration — current iteration (for time calculation)
 */
export function recordMatchEvent(tracker, side, playerID, eventType, detail = {}, iteration = 0) {
  const teamTracker = side === 'home' ? tracker.home : tracker.away;
  const playerStats = getOrCreatePlayerStats(teamTracker, playerID);
  const teamStats = tracker.teamStats[side];
  
  const minute = Math.floor(iteration / 30); // approximate minute
  
  switch (eventType) {
    // ===== SHOOTING =====
    case 'shot':
    case 'shoot':
      playerStats.shots++;
      teamStats.shots++;
      if (detail.onTarget) {
        playerStats.shotsOnTarget++;
        teamStats.shotsOnTarget++;
      } else if (detail.blocked) {
        playerStats.shotsBlocked++;
      } else {
        playerStats.shotsOffTarget++;
      }
      break;
    
    case 'goal':
      playerStats.goals++;
      if (detail.assistPlayerID) {
        const assistPlayer = getOrCreatePlayerStats(teamTracker, detail.assistPlayerID);
        assistPlayer.assists++;
      }
      // Track conceding goalkeeper
      trackConcedingGK(tracker, side, detail);
      break;
    
    // ===== PASSING =====
    case 'pass':
      playerStats.passesAttempted++;
      if (detail.completed) {
        playerStats.passesCompleted++;
      }
      if (detail.keyPass) {
        playerStats.keyPasses++;
      }
      break;
    
    case 'throughBall':
      playerStats.throughBalls++;
      if (detail.completed) {
        playerStats.throughBallsCompleted++;
      }
      break;
    
    case 'cross':
      playerStats.crossesAttempted++;
      if (detail.completed) {
        playerStats.crossesCompleted++;
      }
      break;
    
    // ===== DEFENDING =====
    case 'tackle':
      playerStats.tacklesAttempted++;
      if (detail.won) {
        playerStats.tacklesWon++;
      }
      break;
    
    case 'intercept':
    case 'interception':
      playerStats.interceptions++;
      break;
    
    case 'slide':
      playerStats.tacklesAttempted++;
      if (detail.won) {
        playerStats.tacklesWon++;
      }
      if (detail.foul) {
        playerStats.fouls++;
        teamStats.fouls++;
      }
      break;
    
    case 'cleared':
      playerStats.clearances++;
      break;
    
    // ===== MOVEMENT =====
    case 'sprint':
      playerStats.sprints++;
      playerStats.totalDistance += detail.distance || 5;
      break;
    
    case 'run':
    case 'dribble':
      playerStats.dribblesAttempted++;
      if (detail.completed) {
        playerStats.dribblesCompleted++;
      }
      playerStats.totalDistance += detail.distance || 2;
      break;
    
    // ===== DISCIPLINE =====
    case 'foul':
      playerStats.fouls++;
      teamStats.fouls++;
      if (detail.victimID) {
        const victimTracker = side === 'home' ? tracker.away : tracker.home;
        const victimStats = getOrCreatePlayerStats(victimTracker, detail.victimID);
        victimStats.foulsAgainst++;
      }
      break;
    
    case 'yellowCard':
      playerStats.yellowCards++;
      teamStats.cards++;
      break;
    
    case 'redCard':
      playerStats.redCards++;
      teamStats.cards++;
      break;
    
    // ===== GOALKEEPER =====
    case 'save':
      playerStats.saves++;
      if (detail.parried) {
        playerStats.savesParried++;
      }
      break;
    
    case 'catch':
      playerStats.catches++;
      break;
    
    case 'punch':
      playerStats.punches++;
      break;
    
    case 'sweep':
      playerStats.sweeps++;
      break;
    
    // ===== OTHER =====
    case 'offside':
      playerStats.offsides++;
      break;
    
    case 'error':
      playerStats.positionErrors++;
      if (detail.ledToGoal) {
        playerStats.errorsLeadingToGoal++;
      }
      break;
    
    case 'corner':
      teamStats.corners++;
      break;
  }
  
  // Record key events in timeline
  const keyEvents = ['goal', 'yellowCard', 'redCard', 'substitution'];
  if (keyEvents.includes(eventType)) {
    tracker.matchEvents.push({
      minute,
      iteration,
      side,
      playerID,
      eventType,
      detail,
    });
  }
}

function trackConcedingGK(tracker, scoringSide, detail) {
  // The opposing GK conceded
  const gkSide = scoringSide === 'home' ? 'away' : 'home';
  const gkTracker = gkSide === 'home' ? tracker.home : tracker.away;
  
  // Find GK in team stats
  for (const [playerID, stats] of Object.entries(gkTracker.players)) {
    // We track all keepers — in practice there's only one active GK
    stats.goalsConceded++;
  }
}
```

```javascript
/**
 * Calculate derived statistics after the match.

export function calculateDerivedStats(playerStats) {
  return {
    ...playerStats,
    passAccuracy: playerStats.passesAttempted > 0
      ? (playerStats.passesCompleted / playerStats.passesAttempted * 100).toFixed(1)
      : 0,
    tackleSuccessRate: playerStats.tacklesAttempted > 0
      ? (playerStats.tacklesWon / playerStats.tacklesAttempted * 100).toFixed(1)
      : 0,
    dribbleSuccessRate: playerStats.dribblesAttempted > 0
      ? (playerStats.dribblesCompleted / playerStats.dribblesAttempted * 100).toFixed(1)
      : 0,
    shotAccuracy: playerStats.shots > 0
      ? (playerStats.shotsOnTarget / playerStats.shots * 100).toFixed(1)
      : 0,
    crossAccuracy: playerStats.crossesAttempted > 0
      ? (playerStats.crossesCompleted / playerStats.crossesAttempted * 100).toFixed(1)
      : 0,
  };
}

/**
 * Calculate team-level derived statistics.
 */
export function calculateTeamDerivedStats(teamStats) {
  const totalPasses = teamStats.players 
    ? Object.values(teamStats.players).reduce((sum, p) => sum + (p.passesAttempted || 0), 0)
    : 0;
  const totalTackles = teamStats.players
    ? Object.values(teamStats.players).reduce((sum, p) => sum + (p.tacklesAttempted || 0), 0)
    : 0;
  
  return {
    ...teamStats.total,
    totalPasses,
    totalTackles,
    // Possession will be calculated from pass counts
    // ... more
  };
}
```
