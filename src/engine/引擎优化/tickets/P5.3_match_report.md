```jsx
function MatchOverviewPanel({ matchDetails, statsReport }) {
  const homeStats = statsReport.teamStats.home;
  const awayStats = statsReport.teamStats.away;
  
  // Calculate possession from pass counts
  const totalPasses = homeStats.totalPasses + awayStats.totalPasses;
  const homePossession = totalPasses > 0 
    ? Math.round(homeStats.totalPasses / totalPasses * 100) 
    : 50;
  
  return (
    <div className="match-overview">
      <div className="scoreline">
        <span className="home-score">{matchDetails.goals || 0}</span>
        <span className="separator">-</span>
        <span className="away-score">{matchDetails.awayGoals || 0}</span>
      </div>
      
      <div className="stats-grid">
        <StatBar 
          label="控球率" 
          home={homePossession} 
          away={100 - homePossession}
          format={v => `${v}%`}
        />
        <StatBar 
          label="射门" 
          home={homeStats.shots} 
          away={awayStats.shots} 
        />
        <StatBar 
          label="射正" 
          home={homeStats.shotsOnTarget} 
          away={awayStats.shotsOnTarget} 
        />
        <StatBar 
          label="传球成功率" 
          home={calcTeamPassAccuracy(statsReport.home)} 
          away={calcTeamPassAccuracy(statsReport.away)}
          format={v => `${v}%`}
        />
        <StatBar 
          label="抢断" 
          home={homeStats.totalTackles} 
          away={awayStats.totalTackles} 
        />
        <StatBar 
          label="犯规" 
          home={homeStats.fouls} 
          away={awayStats.fouls} 
        />
        <StatBar 
          label="角球" 
          home={homeStats.corners} 
          away={awayStats.corners} 
        />
      </div>
    </div>
  );
}

function StatBar({ label, home, away, format = v => v }) {
  const total = home + away || 1;
  const homePct = Math.round(home / total * 100);
  
  return (
    <div className="stat-bar-row">
      <span className="stat-value left">{format(home)}</span>
      <div className="stat-bar-label">{label}</div>
      <span className="stat-value right">{format(away)}</span>
      <div className="stat-bar-track">
        <div className="stat-bar-fill home" style={{ width: `${homePct}%` }} />
      </div>
    </div>
  );
}
```

```jsx
function PlayerRatingsTable({ teamName, players, ratings, isHome, formation }) {
  const sorted = [...players]
    .map(p => ({
      ...p,
      rating: ratings[p.playerID] || 6.5,
      stats: p.matchStats || {},
    }))
    .sort((a, b) => b.rating - a.rating);
  
  const bestPlayer = sorted[0];
  
  return (
    <div className="player-ratings">
      <div className="team-header">
        <h3>{teamName}</h3>
        <span className="formation-tag">{formation}</span>
      </div>
      
      {/* Best player highlight */}
      {bestPlayer && (
        <div className="best-player">
          <div className="best-player-rating">{bestPlayer.rating.toFixed(1)}</div>
          <div className="best-player-name">{bestPlayer.name}</div>
          <div className="best-player-pos">{bestPlayer.position}</div>
        </div>
      )}
      
      {/* Full player list */}
      <table className="ratings-table">
        <thead>
          <tr>
            <th>#</th>
            <th>姓名</th>
            <th>位置</th>
            <th>角色</th>
            <th>评分</th>
            <th>进球</th>
            <th>助攻</th>
            <th>传球</th>
            <th>抢断</th>
            <th>跑动(km)</th>
          </tr>
        </thead>
        <tbody>
          {sorted.map((player, idx) => (
            <tr key={player.playerID} className={player.substitute ? 'sub' : ''}>
              <td>{player.shirtNumber || idx + 1}</td>
              <td className="player-name-cell">
                {player.name}
                {player.substitute && <span className="sub-badge">(替补)</span>}
                {player.yellowCards > 0 && <span className="card-icon yellow">🟨</span>}
                {player.redCards > 0 && <span className="card-icon red">🟥</span>}
              </td>
              <td>{player.position}</td>
              <td className="role-cell">{getRoleName(player.role)}</td>
              <td className={`rating-cell ${getRatingClass(player.rating)}`}>
                {player.rating.toFixed(1)}
              </td>
              <td>{player.stats.goals || 0}</td>
              <td>{player.stats.assists || 0}</td>
              <td>{player.stats.passAccuracy || 0}%</td>
              <td>{player.stats.tacklesWon || 0}</td>
              <td>{((player.stats.totalDistance || 0) / 100).toFixed(1)}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

function getRatingClass(rating) {
  if (rating >= 8.0) return 'rating-excellent';
  if (rating >= 7.0) return 'rating-good';
  if (rating >= 6.0) return 'rating-average';
  return 'rating-poor';
}

function getRoleName(roleKey) {
  if (!roleKey) return '-';
  const role = getRoleModifier(roleKey);
  return role?.name || roleKey;
}
```

```jsx
function MatchTimeline({ events, homeTeamName, awayTeamName }) {
  return (
    <div className="match-timeline">
      <h3>比赛事件</h3>
      <div className="timeline-track">
        {events.map((event, idx) => (
          <div key={idx} className={`timeline-event ${event.eventType}`}>
            <span className="event-minute">{event.minute}'</span>
            <span className="event-icon">{getEventIcon(event.eventType)}</span>
            <span className="event-description">
              {formatEvent(event, homeTeamName, awayTeamName)}
            </span>
          </div>
        ))}
      </div>
    </div>
  );
}

function getEventIcon(eventType) {
  const icons = {
    goal: '⚽',
    yellowCard: '🟨',
    redCard: '🟥',
    substitution: '🔄',
  };
  return icons[eventType] || '•';
}

function formatEvent(event, homeTeam, awayTeam) {
  switch (event.eventType) {
    case 'goal':
      return `${event.detail.scorerName || event.playerID} 进球！${event.detail.assistPlayerID ? ` (助攻: ${event.detail.assistPlayerID})` : ''}`;
    case 'yellowCard':
      return `${event.playerID} 黄牌`;
    case 'redCard':
      return `${event.playerID} 红牌！`;
    case 'substitution':
      return `${event.detail.playerInName} ⬆ 换下 ${event.detail.playerOutName}`;
    default:
      return `${event.eventType}: ${event.playerID}`;
  }
}
```

```jsx
            onClick={() => setActiveTab('timeline')}>事件</button>
        </div>
      </div>

        {activeTab === 'overview' && (
          <MatchOverviewPanel matchDetails={matchDetails} statsReport={statsReport} />

        {activeTab === 'home' && (
          <PlayerRatingsTable 
            teamName={matchDetails._homeTeamName}
            players={matchDetails.homePlayers}
            ratings={statsReport.ratings.home}
            isHome={true}
            formation={matchDetails._homeFormation}

        )}

          <PlayerRatingsTable
```

5. **构建完整报告函数**（在 matchEngine.js 中）:

```javascript
/**
 * Build the complete match report from all tracked data.
 * Called after the match ends.
 */
export function buildMatchReport(matchDetails) {
  const tracker = matchDetails._statsTracker;
  if (!tracker) return null;
  
  // Calculate ratings for all players
  const homeRatings = {};
  const awayRatings = {};
  
  // Gather player stats
  for (const [playerID, stats] of Object.entries(tracker.home.players)) {
    const player = findPlayerInMatch(matchDetails, 'home', playerID);
    if (player) {
      const derived = calculateDerivedStats(stats);
      const role = matchDetails._homeRoles?.[playerID] || getDefaultRole(player.position);
      homeRatings[playerID] = calculateRating(
        { ...player, matchStats: derived },
        player.position,
        role,
        { result: getTeamResult(matchDetails, 'home') }
      );
    }
  }
  
  for (const [playerID, stats] of Object.entries(tracker.away.players)) {
    const player = findPlayerInMatch(matchDetails, 'away', playerID);
    if (player) {
      const derived = calculateDerivedStats(stats);
      const role = matchDetails._awayRoles?.[playerID] || getDefaultRole(player.position);
      awayRatings[playerID] = calculateRating(
        { ...player, matchStats: derived },
        player.position,
        role,
        { result: getTeamResult(matchDetails, 'away') }
      );
    }
  }
  
  // Build substitution report
  const subReport = extractSubstitutionReport(
    matchDetails, 
    matchDetails._subTrackerHome, 
    matchDetails._subTrackerAway
  );
  
  return {
    ratings: { home: homeRatings, away: awayRatings },
    teamStats: tracker.teamStats,
    playerStats: { home: tracker.home.players, away: tracker.away.players },
    matchEvents: tracker.matchEvents,
    substitutionReport: subReport,
    formations: {
      home: matchDetails._homeFormation,
      away: matchDetails._awayFormation,
    },
    strategies: {
      home: matchDetails._homeStrategy,
      away: matchDetails._awayStrategy,
    },
  };
}
```
