## 描述

在 `matchEngine.js` 中实现 `applyFormationChange()` 函数，允许在比赛暂停期间切换阵型。切换后所有球员的 originPOS 和 intentPOS 根据新阵型重新计算。

## 背景

FM中的赛中变阵是核心战术操作——从4-4-2切换为4-3-3追分，或从4-3-3变5-3-2守住领先。当前引擎完全不支持此操作，因为 originPOS 在初始化后不再改变。

## 实现内容

### 文件: `D:\newgame\career-sim-react\src\matchEngine.js`

```javascript
/**
 * Apply a formation change during a match (between iterations).
 * Updates all players' originPOS and resets intentPOS based on the new formation.
 *
 * @param {object} matchDetails — current match state
 * @param {'home'|'away'} side — which team is changing formation
 * @param {string} newFormation — formation key (e.g. '4-3-3')
 * @param {object} [pitchSize] — optional, defaults to DEFAULT_PITCH
 * @returns {object} updated matchDetails
 */
export function applyFormationChange(matchDetails, side, newFormation, pitchSize) {
  const p = pitchSize || DEFAULT_PITCH;
  
  // 1. Determine which teams to update
  const teamKey = (side === 'home' && matchDetails.kickOffTeam?.name === matchDetails._homeTeamName)
    ? 'kickOffTeam'
    : (side === 'home' ? 'secondTeam' : 
       matchDetails.kickOffTeam?.name === matchDetails._homeTeamName ? 'secondTeam' : 'kickOffTeam');
  
  const team = matchDetails[teamKey];
  if (!team) return matchDetails;
  
  // 2. Compute new originPOS from formation
  const formationPos = computeOriginPOSForStarters(
    team.players.filter(p => p.currentPOS[0] !== 'NP'), // exclude sent-off
    newFormation,
    p
  );
  
  // 3. Apply new originPOS to each player
  let posIdx = 0;
  for (const player of team.players) {
    if (player.currentPOS[0] === 'NP') continue; // skip sent-off players
    
    player.originPOS = formationPos[posIdx];
    // Reset intentPOS to new originPOS (player will gradually move there)
    player.intentPOS = [...formationPos[posIdx]];
    
    // Don't reset currentPOS — player stays where they are and gradually
    // moves to new intentPOS through formationCheck pull
    
    posIdx++;
  }
  
  // 4. Store formation tag
  if (side === 'home') {
    matchDetails._homeFormation = newFormation;
  } else {
    matchDetails._awayFormation = newFormation;
  }
  
  // 5. Log the change
  matchDetails.iterationLog.push(
    `Formation change: ${team.name} switches to ${newFormation}`
  );
  
  return matchDetails;
}
```

2. **formationCheck强化**:
   - 当 `originPOS !== intentPOS` 时，球员通过 `formationCheck()` 逐渐移向新位置
   - 移动速度合理——不会瞬间传送，而是在几秒的游戏时间内平滑移动
   - 确保 `currentPOS` 路径不会穿过球场墙
