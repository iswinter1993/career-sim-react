2. **球员角色选择器**:

```jsx
function PlayerRoleSelector({ player, value, onChange }) {
  const availableRoles = useMemo(
    () => getAvailableRolesForPosition(player.position),
    [player.position]
  );
  
  const roleDetails = useMemo(() => {
    return availableRoles.map(key => ({
      key,
      ...PLAYER_ROLES[key],
    }));
  }, [availableRoles]);
  
  return (
    <div className="role-selector">
      <div className="player-info">
        <span className="player-pos">{player.position}</span>
        <span className="player-name">{player.name}</span>
      </div>
      <select 
        value={value || getDefaultRole(player.position)}
        onChange={e => onChange(player.playerID, e.target.value)}
      >
        {roleDetails.map(role => (
          <option key={role.key} value={role.key}>
            {role.name} ({role.duty})
          </option>
        ))}
      </select>
    </div>
  );
}
```

3. **完整战术配置面板布局**:

```jsx
function TacticsConfigPanel({ team, onTacticsConfirm }) {
  const [formation, setFormation] = useState('4-4-2');
  const [style, setStyle] = useState('balanced');
  const [mentality, setMentality] = useState('balanced');
  const [roles, setRoles] = useState({}); // {playerID: roleKey}
  
  const handleConfirm = () => {
    onTacticsConfirm({
      formation,
      style: STYLE_PRESETS[style] || style,
      mentality,
      roles, // {playerID: roleKey}
    });
  };
  
  return (
    <div className="tactics-config">
      {/* Row 1: Formation selector (existing from P2.4) */}
      <FormationSelector value={formation} onChange={setFormation} />
      
      {/* Row 2: Team style + Mentality */}
      <div className="tactics-row">
        <TeamStyleSelector value={style} onChange={setStyle} />
        <MentalitySelector value={mentality} onChange={setMentality} />
      </div>
      
      {/* Row 3: Player role assignments */}
      <div className="player-roles">
        <h3>球员角色</h3>
        {team.starters.map(player => (
          <PlayerRoleSelector
            key={player.playerID}
            player={player}
            value={roles[player.playerID]}
            onChange={(pid, role) => setRoles(prev => ({ ...prev, [pid]: role }))}
          />
        ))}
      </div>
      
      <button className="confirm-btn" onClick={handleConfirm}>
        确认战术
      </button>
    </div>
  );
}
```
