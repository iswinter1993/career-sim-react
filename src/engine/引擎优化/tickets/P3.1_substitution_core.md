## 描述

实现完整的 FM 级别换人系统，包括：
1. originPOS 正确继承（替换球员站原位）
2. 3窗口/5人规则
3. 7人替补席管理
4. 换人状态追踪

## 背景

FM的换人规则：
- 正式比赛允许5个换人名额（3次换人窗口，不含中场休息）
- 替补席7人（GK×1 + 外场×6）
- 换人后球员继承被换下球员的阵型站位（originPOS）
- 罚下球员不能通过换人恢复（10人→10人，永久性）
- 有细微的位置覆盖逻辑——如果换人前后位置不匹配，替补球员可能站不同位置

```javascript
// 换人窗口追踪
export function createSubstitutionTracker() {
  return {
    windowsUsed: 0,     // 已用换人窗口数
    maxWindows: 3,      // 最多3个窗口（FIFA规则）
    playersUsed: 0,     // 已用换人名额数
    maxPlayers: 5,      // 最多5人
    substitutions: [],  // [{minute, playerOut, playerIn, window}]
    subbedOutPlayerIDs: new Set(),
  };
}

/**
 * FM-style substitution with originPOS inheritance and substitution rules.
 *
 * @param {object} matchDetails — current match state
 * @param {'home'|'away'} side — which team is substituting
 * @param {string} playerOutID — ID of player to be replaced
 * @param {object} playerIn — squad player object for the substitute
 * @param {object} subTracker — substitution tracker from createSubstitutionTracker()
 * @param {number} minute — current match minute
 * @returns {{matchDetails, subTracker, success: boolean, error?: string}}
 */
export function applySubstitutionV2(matchDetails, side, playerOutID, playerIn, subTracker, minute) {
  // 1. Validate: check substitution limits
  if (subTracker.playersUsed >= subTracker.maxPlayers) {
    return { matchDetails, subTracker, success: false, error: 'MAX_SUBS_REACHED' };
  }
  
  // 2. Validate: playerOut still on pitch
  const teamKey = getTeamKey(matchDetails, side);
  const team = matchDetails[teamKey];
  const outIdx = team.players.findIndex(p => p.playerID === playerOutID);
  if (outIdx === -1) {
    return { matchDetails, subTracker, success: false, error: 'PLAYER_NOT_FOUND' };
  }
  if (team.players[outIdx].currentPOS[0] === 'NP') {
    return { matchDetails, subTracker, success: false, error: 'PLAYER_ALREADY_SENT_OFF' };
  }
  
  // 3. Validate: new substitution window?
  const isNewWindow = subTracker.windowsUsed === 0 || 
    subTracker.substitutions.length === 0 ||
    (minute - subTracker.substitutions.at(-1).minute > 10); // simplified window logic
  
  if (isNewWindow && subTracker.windowsUsed >= subTracker.maxWindows) {
    return { matchDetails, subTracker, success: false, error: 'MAX_WINDOWS_REACHED' };
  }
  
  // 4. originPOS inheritance — substitute stands where the outgoing player stood
  const outPlayer = team.players[outIdx];
  const inheritedOriginPOS = [...outPlayer.originPOS];
  const inheritedCurrentPOS = [...outPlayer.currentPOS];
  
  // 5. Build substitute player JSON
  const playerInWithPos = buildPlayerJson(playerIn, inheritedOriginPOS, {
    isSubstitute: true,
    subbedOnMinute: minute,
    replacedPlayerID: playerOutID,
  });
  
  // Override currentPOS so sub appears at the right spot immediately
  playerInWithPos.currentPOS = inheritedCurrentPOS;
  
  // 6. Swap
  team.players[outIdx] = playerInWithPos;
  
  // 7. Track
  subTracker.playersUsed++;
  if (isNewWindow) subTracker.windowsUsed++;
  subTracker.subbedOutPlayerIDs.add(playerOutID);
  subTracker.substitutions.push({
    minute,
    playerOutID,
    playerInID: playerIn.playerID,
    playerOutName: playerOutID,
    playerInName: playerIn.name,
    window: subTracker.windowsUsed,
  });
  
  // 8. Log
  matchDetails.iterationLog.push(
    `SUB: ${playerIn.name} replaces ${outPlayer.name} (${subTracker.playersUsed}/${subTracker.maxPlayers} subs used)`
  );
  
  return { matchDetails, subTracker, success: true };
}
```

   ```javascript
     const homeTeamName = matchDetails._homeTeamName;
     return (side === 'home' && matchDetails.kickOffTeam?.name === homeTeamName) ||
            (side === 'away' && matchDetails.secondTeam?.name === homeTeamName)
   ```
