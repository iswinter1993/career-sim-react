1. **extractSubstitutionReport()**:

   ```javascript
   /**
    * Generate a substitution report from match details and substitution tracker.
    * Called after match completion.
    */
   export function extractSubstitutionReport(matchDetails, homeSubTracker, awaySubTracker) {
     return {
       home: buildTeamSubReport(matchDetails, 'home', homeSubTracker),
       away: buildTeamSubReport(matchDetails, 'away', awaySubTracker),
     };
   }
   
   function buildTeamSubReport(matchDetails, side, tracker) {
     return {
       totalSubs: tracker.playersUsed,
       windowsUsed: tracker.windowsUsed,
       subs: tracker.substitutions.map(sub => ({
         minute: sub.minute,
         playerOut: sub.playerOutName,
         playerIn: sub.playerInName,
         window: sub.window,
         // Post-sub impact stats (filled from match stats)
         playerInStats: extractPlayerMatchStats(matchDetails, sub.playerInID),
       })),
       remainedUnused: tracker.maxPlayers - tracker.playersUsed,
     };
   }
   ```

2. **在赛后评分中集成换人评价**:

   ```javascript
   /**
    * Rate a substitution decision (for post-match analysis).
    * Considers: timing, positional fit, post-sub goal impact.
    */
   export function rateSubstitution(sub, matchEvents) {
     let score = 5.0; // baseline
     
     // Earlier subs (< 60') score higher — more time to impact
     if (sub.minute < 60) score += 1.0;
     else if (sub.minute < 75) score += 0.5;
     
     // Sub made after conceding a goal → tactical response
     const goalsConcededBefore = matchEvents.filter(
       e => e.type === 'goal' && e.minute < sub.minute
     ).length;
     if (goalsConcededBefore > 0) score += 0.5;
     
     // Sub resulted in a goal (from substitute player)
     const postSubGoals = matchEvents.filter(
       e => e.type === 'goal' && e.minute > sub.minute && 
            (e.scorer === sub.playerInID || e.assist === sub.playerInID)
     ).length;
     score += postSubGoals * 2.0;
     
     // Penalty: sub made right before conceding (tactical error)
     const goalConcededSoon = matchEvents.filter(
       e => e.type === 'goal' && e.minute > sub.minute && e.minute < sub.minute + 5
     ).length;
     score -= goalConcededSoon * 1.5;
     
     return Math.min(10, Math.max(1, score));
   }
   ```
