   ```javascript
   export function buildPlayerJsonList(players, formation, pitchSize) {
     resetPlacementCounters();
     if (formation) {
       const originPOSList = computeOriginPOSForStarters(players, formation, pitchSize);
       return players.map((p, i) => buildPlayerJson(p, originPOSList[i]));
   ```

   ```javascript
   export function applySubstitution(matchDetails, teamKey, playerOutID, playerIn) {
     const idx = team.players.findIndex(p => p.playerID === playerOutID);
     const outOriginPOS = [...team.players[idx].originPOS];
     const subPlayer = buildPlayerJson(playerIn, outOriginPOS);
     team.players[idx] = subPlayer;
     return matchDetails;
   }
   ```
