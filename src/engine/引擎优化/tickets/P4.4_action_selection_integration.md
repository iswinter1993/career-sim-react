## 描述

将 TeamStrategy + PlayerRole + PlayerTraits 三层AI系统集成到 `actions.js` 的动作权重计算中，实现 FM 的 Action Score Equation：

```
Event Score = PlayerGrade + PositionAbility + M(L) + AC
```

其中:
- PlayerGrade → 球员技能值（已有）
- PositionAbility → 位置+角色的加权能力（新增：角色modifiers）
- M(L) → 移动优化因子（已有：formationCheck pull）
- AC → 动作专有能力（已有：基础的action weight）

本ticket的核心是将角色和特质修改器注入现有动作权重计算。

1. **创建 `calculateModifiedActionWeights()` 函数**:

```javascript
import { getRoleModifier, evaluateTraits } from './tactics.js';

/**
 * Calculate the final action weight for a player considering all three AI layers.
 * Replaces the raw position-based weight calculation.
 *
 * @param {object} player — player object with role, traits, skills
 * @param {string} actionName — action key (shoot, pass, tackle, etc.)
 * @param {object} matchState — current match context
 * @param {object} teamStrategy — team-level strategy parameters
 * @returns {number} — final action weight multiplier
 */
export function calculateModifiedActionWeight(player, actionName, matchState, teamStrategy) {
  // 1. Base weight from position (existing logic, already refactored in P1.1)
  const baseWeight = 1.0;
  
  // 2. Role modifier (Layer 2: PlayerRole)
  const role = getRoleModifier(player.role || getDefaultRole(player.position));
  const roleModifier = role.actionModifiers?.[actionName] || 1.0;
  
  // 3. Trait modifier (Layer 3: PlayerTraits)
  const traitEffects = evaluateTraits(player, matchState);
  const traitModifier = 1.0 + (traitEffects.actionModifiers?.[actionName] || 0);
  
  // 4. Team strategy modifier (Layer 1: TeamStrategy)
  let strategyModifier = 1.0;
  if (teamStrategy) {
    if (actionName === 'pass') {
      strategyModifier *= (teamStrategy._passingWeights?.short || 1.0);
    }
    if (actionName === 'throughBall') {
      strategyModifier *= (teamStrategy._passingWeights?.through || 1.0);
    }
    if (actionName === 'shoot' && teamStrategy._tempoMultiplier) {
      strategyModifier *= teamStrategy._tempoMultiplier.pace || 1.0;
    }
    if (actionName === 'sprint') {
      strategyModifier *= (teamStrategy._tempoMultiplier?.pace || 1.0);
    }
    if (actionName === 'tackle' || actionName === 'slide') {
      strategyModifier *= (teamStrategy._pressingMultiplier || 1.0);
    }
  }
  
  // 5. Final weight = base × role × trait × strategy
  const finalMultiplier = roleModifier * traitModifier * strategyModifier;
  
  // 6. Apply to action probability
  return finalMultiplier;
}

/**
 * Generate modified action weights for all actions for a given player.
 * Call this before selecting an action.
 *
 * @returns {{[actionName]: number}} — action weight map
 */
export function getModifiedActionWeights(player, matchState, teamStrategy) {
  const weights = {};
  const actionNames = ['shoot', 'throughBall', 'pass', 'cross', 'tackle', 
                       'intercept', 'slide', 'run', 'sprint', 'cleared', 'boot'];
  
  for (const action of actionNames) {
    weights[action] = calculateModifiedActionWeight(player, action, matchState, teamStrategy);
  }
  
  return weights;
}
```
