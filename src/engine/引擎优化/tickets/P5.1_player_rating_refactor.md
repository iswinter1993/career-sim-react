## 描述

重构 `playerRating.js` 的赛后评分算法，支持全部12个位置和基于球员角色的差异化评分标准。FM的赛后评分基于：球员完成了多少该位置/角色应有的动作。

## 背景

当前评分系统的问题：
1. 只覆盖8个位置（缺少CDM/CAM/LW/RW）
2. 评分逻辑过于简单——所有非GK位置几乎相同的评分标准
3. 没有考虑球员角色的期望行为差异

FM改进方向:
- **B2B中场**: 跑动距离、传球次数、抢断次数权重都高
- **站桩中锋**: 头球成功、关键传球、射门次数权重高
- **内锋**: 盘带成功、内切射门、传球次数权重高
- **清道夫门将**: 出击次数、拦截次数权重高

4. **重构 `calculateRating()` 函数**:

```javascript
/**
 * Calculate FM-style rating for a player after a match.
 *
 * @param {object} player — player with match stats
 * @param {string} position — player's assigned position
 * @param {string} [role] — player's role (optional, uses default if missing)
 * @param {object} matchContext — {scoreline, possession, result}
 * @returns {number} — rating 0.0-10.0
 */
export function calculateRating(player, position, role, matchContext) {
  // 1. Get role-specific weights
  const weights = role ? (ROLE_RATING_WEIGHTS[role] || DEFAULT_RATING_WEIGHTS) : DEFAULT_RATING_WEIGHTS;
  
  // 2. Start from position baseline
  let rating = POSITION_RATING_BASELINE[position] || 6.5;
  
  // 3. Calculate weighted contribution score
  let contributionScore = 0;
  const stats = player.matchStats || {};
  
  for (const [statName, weight] of Object.entries(weights)) {
    const statValue = stats[statName] || 0;
    contributionScore += statValue * weight * 0.1; // 0.1 scaling factor
  }
  
  // 4. Apply mins-played modifier (subs play fewer minutes, don't penalize as hard)
  const minutesPlayed = stats.minutesPlayed || 90;
  const minutesModifier = Math.min(1.0, minutesPlayed / 60); // full rating after 60 mins
  
  // 5. Apply result modifier
  const resultModifier = getResultModifier(matchContext);
  
  // 6. Final rating
  rating += (contributionScore * minutesModifier) + resultModifier;
  
  // 7. Clamp to 0-10
  return Math.max(0.1, Math.min(10.0, rating));
}

function getResultModifier(matchContext) {
  if (!matchContext) return 0;
  
  const { result } = matchContext; // 'win' | 'draw' | 'loss'
  
  switch (result) {
    case 'win': return 0.3;
    case 'draw': return 0.0;
    case 'loss': return -0.3;
    default: return 0;
  }
}
```

5. **GK特殊评分逻辑增强**:

```javascript
export function calculateGKRating(player, role, matchContext) {
  const weights = role ? (ROLE_RATING_WEIGHTS[role] || GK_DEFAULT_WEIGHTS) : GK_DEFAULT_WEIGHTS;
  
  // GK-specific formula
  let rating = POSITION_RATING_BASELINE.GK;
  
  const saves = player.matchStats?.saves || 0;
  const goalsConceded = player.matchStats?.goalsConceded || 0;
  const passesCompleted = player.matchStats?.passesCompleted || 0;
  const sweeps = player.matchStats?.sweeps || 0;
  
  // Save ratio is key for GKs
  const saveRatio = goalsConceded > 0 ? saves / (saves + goalsConceded) : 1.0;
  rating += (saveRatio - 0.7) * 3.0; // 0.7 = average save ratio baseline
  
  // Goal concession penalty
  rating -= goalsConceded * 0.4;
  
  // Clean sheet bonus
  if (goalsConceded === 0) rating += 1.0;
  
  // Distribution bonus (for sweeper keepers especially)
  rating += (passesCompleted / 20) * 0.5;
  
  // Sweeper actions
  rating += sweeps * 0.2;
  
  // Result modifier
  rating += getResultModifier(matchContext);
  
  return Math.max(0.1, Math.min(10.0, rating));
}
```
