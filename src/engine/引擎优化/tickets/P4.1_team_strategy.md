## 描述

在 `engine/lib/tactics.js` 中完成 TeamStrategy（球队风格）的实现。这是FM三层AI的第一层——最高级别的球队整体风格，影响所有球员的行为参数。

## 背景

FM的TeamStrategy包含以下核心维度:
- **Tempo (节奏)**: very_slow / slow / balanced / quick / very_quick — 影响传球倾向和决策速度
- **Pressing Intensity (压迫强度)**: much_less / less / balanced / more / much_more — 影响防守球员的压迫距离和抢断频率
- **Defensive Line (防线高度)**: deep / slightly_deep / balanced / slightly_high / high — 影响防线Y坐标
- **Width (宽度利用)**: narrow / slightly_narrow / balanced / slightly_wide / wide — 影响边路球员X偏移
- **Passing Directness (传球直接性)**: much_shorter / shorter / balanced / more_direct / very_direct — 影响传球偏好
- **Counter Press (反压迫)**: yes/no — 丢球后是否立即集体反抢
- **Fluidity (流动度)**: very_rigid / rigid / balanced / fluid / very_fluid — 影响位置互换频率

2. **applyTeamStrategy() — 应用风格到球队**

```javascript
/**
 * Apply team strategy to a team object.
 * Modifies the team's global parameters that affect all player behavior.
 *
 * @param {object} team — team object with players array
 * @param {object} strategy — team strategy configuration
 * @param {object} pitchSize — {x, y} of the pitch
 * @returns {object} team with strategy applied (mutates input)
 */
export function applyTeamStrategy(team, strategy, pitchSize) {
  team._strategy = { ...DEFAULT_STRATEGY, ...strategy };
  
  // Extract strategy values with defaults
  const s = team._strategy;
  
  // 1. Tempo → affects pace/dribbling aggression
  team._tempoMultiplier = getTempoMultiplier(s.tempo);
  
  // 2. Pressing → affects pressing distance, slide tackle frequency
  team._pressingMultiplier = getPressingMultiplier(s.pressingIntensity);
  team._pressingDistance = getPressingDistance(s.pressingIntensity, pitchSize);
  
  // 3. Defensive line Y offset
  team._defensiveLineOffset = getDefensiveLineOffset(s.defensiveLine, pitchSize);
  
  // 4. Width expansion factor
  team._widthMultiplier = getWidthMultiplier(s.width);
  
  // 5. Passing preferences
  team._passingWeights = getPassingWeights(s.passingDirectness);
  
  // 6. Counter-press flag
  team._counterPress = s.counterPress;
  
  // 7. Fluidity → affects position-swapping willingness
  team._fluidityFactor = getFluidityFactor(s.fluidity);
  
  return team;
}

// 参数化函数 — 每个策略维度映射到数值
function getTempoMultiplier(tempo) {
  const map = {
    very_slow:   { pace: 0.85, passing: 1.10 },
    slow:        { pace: 0.92, passing: 1.05 },
    balanced:    { pace: 1.00, passing: 1.00 },
    quick:       { pace: 1.08, passing: 0.95 },
    very_quick:  { pace: 1.15, passing: 0.90 },
  };
  return map[tempo] || map.balanced;
}

function getPressingMultiplier(intensity) {
  const map = {
    much_less:    0.70,
    less:         0.85,
    balanced:     1.00,
    more:         1.15,
    much_more:    1.30,
  };
  return map[intensity] || 1.0;
}

function getPressingDistance(intensity, pitchSize) {
  const map = {
    much_less:    50,
    less:         80,
    balanced:     120,
    more:         160,
    much_more:    200,
  };
  return map[intensity] || 120;
}

function getDefensiveLineOffset(line, pitchSize) {
  const height = pitchSize ? pitchSize.y : 1050;
  const map = {
    deep:          -60,
    slightly_deep: -30,
    balanced:      0,
    slightly_high: 30,
    high:          60,
  };
  return (map[line] || 0) * (height / 1050); // scale to pitch
}

function getWidthMultiplier(width) {
  const map = {
    narrow:         0.85,
    slightly_narrow: 0.93,
    balanced:       1.00,
    slightly_wide:  1.07,
    wide:           1.15,
  };
  return map[width] || 1.0;
}

function getFluidityFactor(fluidity) {
  const map = {
    very_rigid: 0.0,  // no position swapping
    rigid:      0.1,
    balanced:   0.3,
    fluid:      0.5,
    very_fluid: 0.7,
  };
  return map[fluidity] || 0.3;
}
```
